## connect
S RX v
S TX VNexDome V 1.10
S RX y
S TX Y 0
S RX q
S TX Q 90.0
S RX n
S TX N 180.00
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U CONNECTION BUSY CONNECTED
U INFO OK DEVICE_MODEL=NexDome DEVICE_FIRMWARE_REVISION=1.10
D NEXDOME_REVERSED OK RW count=2 NO
D NEXDOME_RESET_SHUTTER_COMM OK RW count=1
D NEXDOME_FIND_HOME OK RW count=1
D NEXDOME_CALLIBRATE OK RW count=1
D NEXDOME_POWER OK RO count=2 ROTATOR_VOLTAGE=0/0 SHUTTER_VOLTAGE=0/0
U DOME_PARK OK UNPARKED
D DOME_DIRECTION OK RW count=2 MOVE_CLOCKWISE
D DOME_ON_COORDINATES_SET OK RW count=1 GOTO
D DOME_STEPS OK RW count=1 STEPS=0/0
D DOME_HORIZONTAL_COORDINATES OK RW count=1 AZ=0/0
D DOME_SLAVING_PARAMETERS OK RW count=1 MOVE_THRESHOLD=1/1
D DOME_ABORT_MOTION OK RW count=1
D DOME_SHUTTER OK RW count=2 CLOSED
D DOME_PARK OK RW count=2 UNPARKED
U CONNECTION OK CONNECTED
U NEXDOME_POWER OK ROTATOR_VOLTAGE=12.6/0 SHUTTER_VOLTAGE=12.5/0
U DOME_HORIZONTAL_COORDINATES OK AZ=90/0
U DOME_STEPS OK STEPS=0/0
U DOME_SHUTTER OK CLOSED
## goto 180
S RX g 180.00
S TX G
S POLL [K 1260 1250 1100] [M 1] [Q #] [U 3 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=0/0
U DOME_HORIZONTAL_COORDINATES OK AZ=180/180
U DOME_STEPS OK STEPS=0/0
## steps clockwise 30
S RX q
S TX Q 180.0
S RX g 210.00
S TX G
S POLL [K 1260 1250 1100] [M 1] [Q #] [U 3 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_DIRECTION OK MOVE_CLOCKWISE
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=30/30
U DOME_HORIZONTAL_COORDINATES OK AZ=210/180
U DOME_STEPS OK STEPS=30/30
## steps counterclockwise 40
S RX q
S TX Q 210.0
S RX g 170.00
S TX G
S POLL [K 1260 1250 1100] [M 1] [Q #] [U 3 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_DIRECTION OK MOVE_COUNTERCLOCKWISE
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=40/40
U DOME_HORIZONTAL_COORDINATES OK AZ=170/180
U DOME_STEPS OK STEPS=40/40
## direction clockwise
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_DIRECTION OK MOVE_CLOCKWISE
## park
S RX n
S TX N 180.00
S RX g 180.00
S TX G
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_STEPS BUSY STEPS=40/40
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_PARK BUSY UNPARKED
U DOME_HORIZONTAL_COORDINATES OK AZ=180/180
U DOME_STEPS OK STEPS=40/40
U DOME_PARK OK PARKED
## goto while parked
S RX q
S TX Q 180.0
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_HORIZONTAL_COORDINATES ALERT AZ=180/90
M DOME_HORIZONTAL_COORDINATES "Dome is parked"
## steps while parked
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_STEPS ALERT STEPS=10/10
M DOME_STEPS "Dome is parked"
## unpark
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_PARK OK UNPARKED
## find home
S RX h
S TX H
S POLL [K 1260 1250 1100] [M 2] [Q #] [U 3 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U NEXDOME_FIND_HOME BUSY FIND_HOME
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=10/10
U DOME_HORIZONTAL_COORDINATES OK AZ=0/90
U DOME_STEPS OK STEPS=10/10
U NEXDOME_FIND_HOME OK
M NEXDOME_FIND_HOME "Home Found."
## calibration
S RX c
S TX C
S POLL [K 1260 1250 1100] [M 3] [Q #] [U 3 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U NEXDOME_CALLIBRATE BUSY CALLIBRATE
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=10/10
U DOME_HORIZONTAL_COORDINATES OK AZ=0/90
U DOME_STEPS OK STEPS=10/10
U NEXDOME_CALLIBRATE OK
M NEXDOME_CALLIBRATE "Callibration complete."
## goto 200
S RX g 200.00
S TX G
S POLL [K 1260 1250 1100] [M 1] [Q #] [U 3 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=10/10
U DOME_HORIZONTAL_COORDINATES OK AZ=200/200
U DOME_STEPS OK STEPS=10/10
## abort in motion
S RX g 100.00
S TX G
S POLL [K # # #] [M #] [Q #] [U # #]
S RX a
S TX A
S POLL [K # # #] [M #] [Q #] [U # #]
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=#/#
U DOME_ABORT_MOTION OK
U DOME_HORIZONTAL_COORDINATES OK AZ=#/#
U DOME_STEPS OK STEPS=#/#
## goto 300 after abort
S RX g 300.00
S TX G
S POLL [K 1260 1250 1100] [M 1] [Q #] [U 3 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_HORIZONTAL_COORDINATES BUSY AZ=*
U DOME_STEPS BUSY STEPS=10/10
U DOME_HORIZONTAL_COORDINATES OK AZ=300/300
U DOME_STEPS OK STEPS=10/10
## abort idle
S RX a
S TX A
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_ABORT_MOTION OK
## failed abort
S RX a
S FAULT a error
S TX E
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_ABORT_MOTION ALERT
## failed goto
S RX g 10.00
S FAULT g 10.00 error
S TX E
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_HORIZONTAL_COORDINATES ALERT AZ=10/10
## shutter open
S RX d
S TX D
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 2 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 1 1]
U DOME_SHUTTER BUSY OPENED
U DOME_SHUTTER OK OPENED
## shutter close
S RX e
S TX D
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 4 1]
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U DOME_SHUTTER BUSY CLOSED
U DOME_SHUTTER BUSY OPENED
U DOME_SHUTTER OK CLOSED
## reversed yes
S RX y 1
S TX Y 1
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U NEXDOME_REVERSED OK YES
## reversed no
S RX y 0
S TX Y 0
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U NEXDOME_REVERSED OK NO
## low voltage
S POLL [K 720 1250 1100] [M 0] [Q #] [U 3 1]
M ALERT "Dome power is low! (U_rotator = 7.20V, U_shutter = 12.50V)"
U NEXDOME_POWER OK ROTATOR_VOLTAGE=7.2/0 SHUTTER_VOLTAGE=12.5/0
## normal voltage
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
M IDLE "Dome power is normal! (U_rotator = 12.60V, U_shutter = 12.50V)"
U NEXDOME_POWER OK ROTATOR_VOLTAGE=12.6/0 SHUTTER_VOLTAGE=12.5/0
## reset shutter communication
S RX w
S TX W
S POLL [K 1260 1250 1100] [M 0] [Q #] [U 3 1]
U NEXDOME_RESET_SHUTTER_COMM BUSY RESET
U NEXDOME_RESET_SHUTTER_COMM OK
## disconnect
U CONNECTION BUSY DISCONNECTED
X NEXDOME_REVERSED
X NEXDOME_RESET_SHUTTER_COMM
X NEXDOME_FIND_HOME
X NEXDOME_CALLIBRATE
X NEXDOME_POWER
X DOME_DIRECTION
X DOME_ON_COORDINATES_SET
X DOME_STEPS
X DOME_HORIZONTAL_COORDINATES
X DOME_SLAVING_PARAMETERS
X DOME_ABORT_MOTION
X DOME_SHUTTER
X DOME_PARK
U CONNECTION OK DISCONNECTED
